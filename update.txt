#!/bin/bash
if [ "$(id -u)" -ne 0 ]; then echo "This requires root permission."; exit 1; fi

mkdir -p /var/lib/scpkgs/installed
chmod 755 /var/lib/scpkgs
chmod 755 /var/lib/scpkgs/installed

if [ ! -f /etc/scpkgs/repos.conf ]; then
    mkdir -p /etc/scpkgs
    cat << 'CONF' > /etc/scpkgs/repos.conf
# scpkgs repository configuration
# Format:
#   repo <name> <raw_base_url> <api_url>
#
# raw_base_url  - direct URL base for downloading .scrap files
# api_url       - GitHub API contents URL for package listing
#
# The first repo listed is the default/preferred repo.
# Add your own repos below the official one.
#
# Example:
#   repo myrepo https://raw.githubusercontent.com/user/myrepo/main https://api.github.com/repos/user/myrepo/contents

repo official https://raw.githubusercontent.com/apiwo/scrap-pkg/main https://api.github.com/repos/apiwo/scrap-pkg/contents
CONF
    echo "Created default repo config at /etc/scpkgs/repos.conf"
fi

MSG="Installed scpkgs."
[ -f /usr/local/bin/scpkgs ] && MSG="Updated scpkgs."

cat << 'EOF' > /usr/local/bin/scpkgs
#!/bin/bash

REPOS_CONF="/etc/scpkgs/repos.conf"
INSTALLED_DB="/var/lib/scpkgs/installed"
SELF="/usr/local/bin/scpkgs"
PROBE_TIMEOUT=2

declare -a REPO_NAMES=()
declare -a REPO_RAW=()
declare -a REPO_API=()

load_repos() {
    if [ ! -f "$REPOS_CONF" ]; then
        REPO_NAMES=("official")
        REPO_RAW=("https://raw.githubusercontent.com/apiwo/scrap-pkg/main")
        REPO_API=("https://api.github.com/repos/apiwo/scrap-pkg/contents")
        return
    fi
    while IFS= read -r line; do
        [[ "$line" =~ ^#.*$ || -z "${line// }" ]] && continue
        if [[ "$line" =~ ^repo[[:space:]]+([^[:space:]]+)[[:space:]]+([^[:space:]]+)[[:space:]]+([^[:space:]]+) ]]; then
            REPO_NAMES+=("${BASH_REMATCH[1]}")
            REPO_RAW+=("${BASH_REMATCH[2]}")
            REPO_API+=("${BASH_REMATCH[3]}")
        fi
    done < "$REPOS_CONF"
    if [ ${#REPO_NAMES[@]} -eq 0 ]; then
        REPO_NAMES=("official")
        REPO_RAW=("https://raw.githubusercontent.com/apiwo/scrap-pkg/main")
        REPO_API=("https://api.github.com/repos/apiwo/scrap-pkg/contents")
    fi
}

load_repos

show_help() {
    echo "scpkgs - Independent Linux Package Manager"
    echo ""
    echo "  Package operations:"
    echo "  <pkg>                      Build and install a package"
    echo "  -n <pkg>                   Install without confirmation"
    echo "  -nodeps <pkg>              Install without checking dependencies"
    echo "  -r <pkg>                   Remove a package"
    echo "  -rr <pkg>                  Force remove (no confirmation)"
    echo "  -d <pkg>                   Download source only"
    echo "  -k <pkg>                   Keep build files after installation"
    echo "  -we <pkg>                  Treat warnings as errors & strict build"
    echo "  -vc <pkg>                  Manually specify a version to install"
    echo "  -vch <pkg>                 Hard override the source URL"
    echo "  -c <url>                   Build/Install directly from a URL"
    echo ""
    echo "  Repository:"
    echo "  -u                         List packages from all repos"
    echo "  -u <reponame>              List packages from a specific repo"
    echo "  -update                    Run the update script from default repo"
    echo "  -repos                     List configured repositories"
    echo "  -repo <name> <pkg>         Install a package from a specific repo"
    echo "  -s <pkg>                   Search for a package across all repos"
    echo "  -i <pkg>                   Get detailed package info from all repos"
    echo ""
    echo "  Misc:"
    echo "  -l                         List all installed packages"
    echo "  makescrap                  Generate a template .scrap file"
}

makescrap_cmd() {
    cat << 'TEMPLATE' > template.scrap
# This is the header of a scrapfile
name="mypkg"
desc="A short description."
info="https://example.com"
version="1.0.0"
license="GPLv2"
maintainer="apiwo"
arch="x86_64"
priority="optional"
depends="gcc, make, libc-dev"
build-type="tarball"
download="https://example.com/mypkg-1.0.0.tar.gz"
build() {
    ./configure --prefix=/usr
    make
}
instructions() {
    make DESTDIR="$SCRAP_PKG_DEST" install
}
TEMPLATE
    echo "Created template.scrap"
    exit 0
}

FETCHED_REPO_NAME=""
FETCHED_REPO_INDEX=0

fetch_scrap() {
    local pkg="$1"
    local preferred="$2"
    local target="/tmp/${pkg}.scrap"

    if [ -n "$preferred" ]; then
        wget -q --user-agent="Mozilla/5.0" "${REPO_RAW[$preferred]}/${pkg}.scrap" -O "$target" 2>/dev/null
        if [ $? -eq 0 ] && [ -s "$target" ]; then
            FETCHED_REPO_NAME="${REPO_NAMES[$preferred]}"
            FETCHED_REPO_INDEX=$preferred
            echo "$target"; return 0
        fi
        wget -q --user-agent="Mozilla/5.0" "${REPO_RAW[$preferred]}/${pkg}.clump" -O "$target" 2>/dev/null
        if [ $? -eq 0 ] && [ -s "$target" ]; then
            FETCHED_REPO_NAME="${REPO_NAMES[$preferred]}"
            FETCHED_REPO_INDEX=$preferred
            echo "$target"; return 0
        fi
        rm -f "$target"; return 1
    fi

    local found_indices=()
    for i in "${!REPO_NAMES[@]}"; do
        wget -q --user-agent="Mozilla/5.0" "${REPO_RAW[$i]}/${pkg}.scrap" -O "$target" 2>/dev/null
        if [ $? -eq 0 ] && [ -s "$target" ]; then found_indices+=("$i"); rm -f "$target"; continue; fi
        wget -q --user-agent="Mozilla/5.0" "${REPO_RAW[$i]}/${pkg}.clump" -O "$target" 2>/dev/null
        if [ $? -eq 0 ] && [ -s "$target" ]; then found_indices+=("$i"); rm -f "$target"; fi
    done

    if [ ${#found_indices[@]} -eq 0 ]; then rm -f "$target"; return 1; fi

    local chosen="${found_indices[0]}"
    if [ ${#found_indices[@]} -gt 1 ]; then
        echo ""
        echo "Package '$pkg' found in multiple repos:"
        for idx in "${found_indices[@]}"; do
            echo "  [$idx] ${REPO_NAMES[$idx]}"
        done
        read -p "Which repo to use? [default: $chosen]: " choice
        if [[ "$choice" =~ ^[0-9]+$ ]]; then
            for idx in "${found_indices[@]}"; do
                [ "$choice" -eq "$idx" ] && chosen=$choice && break
            done
        fi
    fi

    wget -q --user-agent="Mozilla/5.0" "${REPO_RAW[$chosen]}/${pkg}.scrap" -O "$target" 2>/dev/null
    if [ $? -eq 0 ] && [ -s "$target" ]; then
        FETCHED_REPO_NAME="${REPO_NAMES[$chosen]}"
        FETCHED_REPO_INDEX=$chosen
        echo "$target"; return 0
    fi
    wget -q --user-agent="Mozilla/5.0" "${REPO_RAW[$chosen]}/${pkg}.clump" -O "$target" 2>/dev/null
    if [ $? -eq 0 ] && [ -s "$target" ]; then
        FETCHED_REPO_NAME="${REPO_NAMES[$chosen]}"
        FETCHED_REPO_INDEX=$chosen
        echo "$target"; return 0
    fi
    rm -f "$target"; return 1
}

check_dep_installed() {
    local dep="$1"
    [ -f "$INSTALLED_DB/$dep.list" ] && return 0
    command -v "$dep" >/dev/null 2>&1 && return 0

    local base="${dep%-dev}"
    local libname="${base#lib}"

    local header_dirs=(
        "/usr/include" "/usr/local/include"
        "/usr/include/x86_64-linux-gnu"
        "/usr/include/X11" "/usr/include/X11/extensions"
        "/usr/include/GL" "/usr/include/EGL"
        "/usr/include/vulkan" "/usr/include/wayland"
        "/usr/include/freetype2" "/usr/include/harfbuzz"
        "/usr/include/glib-2.0" "/usr/include/dbus-1.0"
    )
    for d in "${header_dirs[@]}"; do
        [ -f "$d/${dep}.h" ]     && return 0
        [ -f "$d/${base}.h" ]    && return 0
        [ -f "$d/${libname}.h" ] && return 0
    done

    local lib_dirs=(
        "/usr/lib" "/usr/lib64" "/usr/lib/x86_64-linux-gnu"
        "/usr/local/lib" "/usr/local/lib64" "/lib" "/lib64"
    )
    for d in "${lib_dirs[@]}"; do
        [ -f "$d/lib${libname}.so" ] && return 0
        [ -f "$d/lib${libname}.a" ]  && return 0
        ls "$d/lib${libname}.so."* >/dev/null 2>&1 && return 0
    done

    if command -v pkg-config >/dev/null 2>&1; then
        pkg-config --exists "$libname" >/dev/null 2>&1 && return 0
        pkg-config --exists "$base"    >/dev/null 2>&1 && return 0
        pkg-config --exists "$dep"     >/dev/null 2>&1 && return 0
    fi

    if command -v gcc >/dev/null 2>&1; then
        echo "int main(){}" | gcc -x c - -l"$libname" -o /dev/null >/dev/null 2>&1 && return 0
    fi

    return 1
}

resolve_deps() {
    local normal_deps="$1"
    local split_deps="$2"

    if [ -n "$normal_deps" ]; then
        IFS=',' read -ra DEPS <<< "$normal_deps"
        for d in "${DEPS[@]}"; do
            d=$(echo "$d" | xargs)
            [ -z "$d" ] && continue
            if ! check_dep_installed "$d"; then
                echo "[!] Missing dependency: $d. Attempting auto-install..."
                bash "$SELF" -n "$d"
                if [ $? -ne 0 ]; then
                    echo "Failed to install dependency: $d"
                    exit 1
                fi
            else
                echo "[ OK ] $d"
            fi
        done
    fi

    if [ -n "$split_deps" ]; then
        IFS=',' read -ra SDEPS <<< "$split_deps"
        for sd in "${SDEPS[@]}"; do
            sd=$(echo "$sd" | xargs)
            [ -z "$sd" ] && continue
            IFS='>' read -ra ALTS <<< "$sd"
            local found=false
            for alt in "${ALTS[@]}"; do
                alt=$(echo "$alt" | xargs)
                if check_dep_installed "$alt"; then
                    echo "[ OK ] $alt (alternative)"
                    found=true; break
                fi
            done
            if [ "$found" = false ]; then
                local preferred
                preferred=$(echo "${ALTS[0]}" | xargs)
                echo "[!] Missing: $sd. Installing preferred: $preferred..."
                bash "$SELF" -n "$preferred" || exit 1
            fi
        done
    fi
}

climb_logic() {
    local cur_v="$1" cur_u="$2" best_v="$1" best_u="$2"
    [ "$CLIMB" == "NO" ] && { FINAL_VER="$cur_v"; FINAL_URL="$cur_u"; return; }
    echo -n "Probing versions from $cur_v: "
    local dots; dots=$(echo "$cur_v" | tr -cd '.' | wc -c)
    while true; do
        local found_step=false
        for lookahead in 1 2 3; do
            local major minor patch next_v major_v next_u major_u
            major=$(echo "$cur_v" | cut -d. -f1)
            minor=$(echo "$cur_v" | cut -d. -f2)
            patch=$(echo "$cur_v" | cut -d. -f3)
            if [ "$dots" -eq 2 ]; then
                next_v="${major}.${minor}.$((patch + lookahead))"; major_v="$((major+1)).0.0"
            else
                next_v="${major}.$((minor + lookahead))"; major_v="$((major+1)).0"
            fi
            next_u="${cur_u//$cur_v/$next_v}"
            major_u="${cur_u//$cur_v/$major_v}"; major_u="${major_u/\/v${major}\//\/v$((major+1))\/}"
            if wget -q --spider -T "$PROBE_TIMEOUT" -t 1 --user-agent="Mozilla/5.0" "$next_u" 2>/dev/null; then
                echo -n "-> $next_v "; cur_v="$next_v"; cur_u="$next_u"; best_v="$next_v"; best_u="$next_u"; found_step=true; break
            elif wget -q --spider -T "$PROBE_TIMEOUT" -t 1 --user-agent="Mozilla/5.0" "$major_u" 2>/dev/null; then
                echo -n "-> $major_v "; cur_v="$major_v"; cur_u="$major_u"; best_v="$major_v"; best_u="$major_u"; found_step=true; break
            fi
        done
        [ "$found_step" = false ] && { echo "(Stable)"; break; }
    done
    FINAL_VER="$best_v"; FINAL_URL="$best_u"
}

# ── ROOTLESS ──────────────────────────────────────────────────────────────────
case "$1" in
    ""|-h|--help) show_help; exit 0 ;;

    -l)
        echo "Installed via scpkgs:"
        echo "---"
        pkgs=$(ls "$INSTALLED_DB" 2>/dev/null | sed 's/\.list$//' | sort)
        if [ -z "$pkgs" ]; then
            echo "  (none)"
        else
            echo "$pkgs" | while IFS= read -r p; do echo "  $p"; done
            echo "---"
            echo "  Total: $(echo "$pkgs" | wc -l) package(s)"
        fi
        exit 0 ;;

    -repos)
        echo "Configured repositories:"
        echo "---"
        for i in "${!REPO_NAMES[@]}"; do
            label=""; [ $i -eq 0 ] && label="  (default)"
            echo "  [${REPO_NAMES[$i]}]$label"
            echo "    raw: ${REPO_RAW[$i]}"
            echo "    api: ${REPO_API[$i]}"
        done
        echo "---"
        echo "Config: $REPOS_CONF"
        exit 0 ;;

    -s)
        [ -z "$2" ] && echo "Usage: scpkgs -s <term>" && exit 1
        echo "Searching all repos for '$2'..."
        echo "---"
        for i in "${!REPO_NAMES[@]}"; do
            echo "  [${REPO_NAMES[$i]}]"
            results=$(curl -sL -H "User-Agent: scpkgs" "${REPO_API[$i]}" \
                | grep -oP '"name":\s*"\K[^"]+' \
                | grep -E '\.(scrap|clump)$' \
                | grep -i "$2" \
                | sed -E 's/\.(scrap|clump)//')
            if [ -n "$results" ]; then
                echo "$results" | while IFS= read -r pkg; do echo "    $pkg"; done
            else
                echo "    (no results)"
            fi
        done
        echo "---"
        exit 0 ;;

    -i)
        [ -z "$2" ] && echo "Usage: scpkgs -i <pkg>" && exit 1
        found_any=false
        for i in "${!REPO_NAMES[@]}"; do
            target="/tmp/$2_info.scrap"
            wget -q --user-agent="Mozilla/5.0" "${REPO_RAW[$i]}/$2.scrap" -O "$target" 2>/dev/null
            if [ $? -eq 0 ] && [ -s "$target" ]; then
                echo "--- [${REPO_NAMES[$i]}] ---"
                grep -E '^(name|version|desc|maintainer|depends|license|arch|priority)=' "$target" | sed 's/"//g'
                rm -f "$target"; found_any=true
            fi
        done
        [ "$found_any" = false ] && echo "Package '$2' not found in any repo."
        exit 0 ;;

    makescrap) makescrap_cmd ;;
esac

KEEP=false; STRICT_ERRORS=false; IS_CLUMP=false; RESOLVE_DEPS=true
PKG=""; FORCED_REPO_INDEX=""

while [[ "$#" -gt 0 ]]; do

        -u)

            if [ -n "$2" ] && [[ "$2" != -* ]]; then
                found_idx=""
                for i in "${!REPO_NAMES[@]}"; do
                    [ "${REPO_NAMES[$i]}" = "$2" ] && found_idx=$i && break
                done
                if [ -z "$found_idx" ]; then
                    echo "Unknown repo '$2'. Known: ${REPO_NAMES[*]}"; exit 1
                fi
                echo "Packages in [${REPO_NAMES[$found_idx]}]:"
                echo "---"
                curl --no-buffer -sL -H "User-Agent: scpkgs" "${REPO_API[$found_idx]}" \
                    | grep -oP '"name":\s*"\K[^"]+' \

# ── RESOLUTION ────────────────────────────────────────────────────────────────
if [ "$DIRECT_URL" = true ]; then
    name="custom"; version="remote"; download="$DOWNLOAD_LINK"; build_type="tarball"
    build()        { ./configure --prefix=/usr && make; }
    instructions() { make DESTDIR="$SCRAP_PKG_DEST" install; }
else
    file=$(fetch_scrap "$PKG" "$FORCED_REPO_INDEX")
    if [ ! -f "$file" ]; then
        echo "Package '$PKG' not found in any configured repository."
        echo "Configured repos:"; for n in "${REPO_NAMES[@]}"; do echo "  $n"; done
        exit 1
    fi
    echo "[ repo: $FETCHED_REPO_NAME ] Found $PKG"
    grep -q "#SCPKG CLUMP" "$file" && IS_CLUMP=true
    sed -i 's/build-type/build_type/g' "$file"
    source "$file"
    if [ "$IS_CLUMP" = false ]; then
        if [ "$HARD_URL" = true ]; then
            read -p "New source URL: " download; version="custom"
        elif [ "$CUSTOM" = true ]; then
            read -p "Target version: " target_v
            m_old=$(echo "$version" | cut -d. -f1); m_new=$(echo "$target_v" | cut -d. -f1)
            download="${download//$version/$target_v}"; download="${download/\/v${m_old}\//\/v${m_new}\/}"
            version="$target_v"
        elif [[ "$build_type" =~ ^(tarball|cmake|meson)$ ]]; then
            climb_logic "$version" "$download"; version="$FINAL_VER"; download="$FINAL_URL"
        fi
    fi
fi

[ "$CONFIRM" != false ] && read -p "Proceed with $name $version? [y/N]: " ans && [[ "$ans" != "y" ]] && exit 0

if [ "$RESOLVE_DEPS" = true ]; then
    echo "Resolving dependencies..."
    resolve_deps "$depends" "$split_depend"
fi

[ "$STRICT_ERRORS" = true ] && { export CFLAGS="$CFLAGS -Werror"; set -e; }

WORK_DIR="/tmp/scpkgs-$name"
export SCRAP_PKG_DEST="/tmp/scpkgs-dest-$name"
rm -rf "$WORK_DIR" "$SCRAP_PKG_DEST"
mkdir -p "$WORK_DIR" "$SCRAP_PKG_DEST"
cd "$WORK_DIR" || { echo "Failed to enter work directory."; exit 1; }

if [ "$IS_CLUMP" = true ]; then
    grep "^source:" "$file" | cut -d':' -f2- | while IFS= read -r src; do
        src=$(echo "$src" | xargs)
        wget -q --show-progress "$src" -O p.tar
        tar -xf p.tar 2>/dev/null || unzip -q p.tar 2>/dev/null; rm -f p.tar
    done
else
    if [ "$build_type" == "git" ] || [[ "$download" == *.git ]]; then
        echo "Cloning $download..."
        git clone --depth=1 --quiet "$download" . || { echo "Git clone failed."; exit 1; }
    else
        echo "Downloading $name $version..."
        wget -q --show-progress "$download" -O src.tar || { echo "Download failed."; exit 1; }
        tar -xf src.tar --strip-components=1 2>/dev/null || tar -xf src.tar 2>/dev/null
        rm -f src.tar
    fi
    case "$build_type" in
        cmake)
            if ! declare -f build | grep -q cmake 2>/dev/null; then
                build()        { mkdir -p b && cd b && cmake .. -DCMAKE_BUILD_TYPE=Release -DCMAKE_INSTALL_PREFIX=/usr && make; }
                instructions() { cd b && make DESTDIR="$SCRAP_PKG_DEST" install; }
            fi ;;
        meson)
            if ! declare -f build | grep -q meson 2>/dev/null; then
                build()        { meson setup b --prefix=/usr && ninja -C b; }
                instructions() { DESTDIR="$SCRAP_PKG_DEST" ninja -C b install; }
            fi ;;
        go)
            if ! declare -f build | grep -q "go build" 2>/dev/null; then
                build()        { go build -o "$name" .; }
                instructions() { mkdir -p "$SCRAP_PKG_DEST/usr/bin" && cp "$name" "$SCRAP_PKG_DEST/usr/bin/"; }
            fi ;;
        rust)
            if ! declare -f build | grep -q cargo 2>/dev/null; then
                build()        { cargo build --release; }
                instructions() { mkdir -p "$SCRAP_PKG_DEST/usr/bin" && cp "target/release/$name" "$SCRAP_PKG_DEST/usr/bin/"; }
            fi ;;
    esac
fi

echo "Building $name..."
build || { echo "Build failed."; exit 1; }
echo "Installing $name..."
instructions || { echo "Install step failed."; exit 1; }

# Deploy to system first, then record — so the DB only reflects what actually landed
cp -r "$SCRAP_PKG_DEST"/* / 2>/dev/null

# Build the file list from what was actually staged
# Use both the scrapfile's internal name AND the PKG name used to call us,
# so -l and -r always find it regardless of capitalisation in the scrapfile.
FILELIST=$(find "$SCRAP_PKG_DEST" -type f | sed "s|^$SCRAP_PKG_DEST||")

# Normalise: always lowercase for the DB key
DB_KEY=$(echo "$name" | tr '[:upper:]' '[:lower:]')
echo "$FILELIST" > "$INSTALLED_DB/${DB_KEY}.list"

# If the caller used a different name (e.g. typed "Nano" or "nano"),
# write a symlink-style alias entry so both spellings resolve.
PKG_KEY=$(echo "$PKG" | tr '[:upper:]' '[:lower:]')
if [ -n "$PKG_KEY" ] && [ "$PKG_KEY" != "$DB_KEY" ]; then
    echo "$FILELIST" > "$INSTALLED_DB/${PKG_KEY}.list"
fi

echo "[ OK ] Installed $name $version."

[ "$KEEP" = false ] && rm -rf "$WORK_DIR" "$SCRAP_PKG_DEST" "$file"
set +e
EOF

chmod +x /usr/local/bin/scpkgs
echo "$MSG"
